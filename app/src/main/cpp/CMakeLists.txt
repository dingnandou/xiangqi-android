cmake_minimum_required(VERSION 3.22.1)
project(phone_chess_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT PIKAFISH_DIR)
    set(PIKAFISH_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../engine/src/main/cpp/pikafish")
endif()
file(TO_CMAKE_PATH "${PIKAFISH_DIR}" PIKAFISH_DIR)
file(GLOB PIKA_ROOT "${PIKAFISH_DIR}/*.cpp")
file(GLOB_RECURSE PIKA_NNUE "${PIKAFISH_DIR}/nnue/*.cpp")
file(GLOB_RECURSE PIKA_EXTERNAL "${PIKAFISH_DIR}/external/*.cpp")
list(FILTER PIKA_ROOT EXCLUDE REGEX "/main\\.cpp$")
add_library(strongengine SHARED strong_bridge.cpp ${PIKA_ROOT} ${PIKA_NNUE} ${PIKA_EXTERNAL})
target_include_directories(strongengine PRIVATE "${PIKAFISH_DIR}")
target_compile_options(strongengine PRIVATE -O3 -fno-exceptions -fvisibility=hidden -ffunction-sections -fdata-sections)
target_compile_definitions(strongengine PRIVATE NDEBUG IS_64BIT ZSTD_DISABLE_ASM)
if(ANDROID_ABI STREQUAL "arm64-v8a")
    target_compile_definitions(strongengine PRIVATE USE_POPCNT USE_NEON=8 ARCH=armv8)
elseif(ANDROID_ABI STREQUAL "x86_64")
    target_compile_definitions(strongengine PRIVATE USE_SSE2 NO_PREFETCH)
endif()
target_link_options(strongengine PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--gc-sections")
